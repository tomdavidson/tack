name: "gh-settings"
description: "Merge org enforced + default GitHub settings with repo overrides, then plan/apply via OpenTofu"

inputs:
  mode:
    required: true
    description: "'plan' or 'apply'"
  repo-settings-path:
    required: false
    default: ".github/repo-settings.yml"
    description: "Path (relative to the calling repo) to the per-repo overrides YAML"
  enforced-path:
    required: false
    default: ""
    description: "Override path to enforced.yml. Defaults to the bundled one in the action."
  defaults-path:
    required: false
    default: ""
    description: "Override path to defaults.yml. Defaults to the bundled one in the action."
  tofu-dir:
    required: false
    default: ""
    description: "Override path to the OpenTofu module. Defaults to the bundled one in the action."
  tofu-version:
    required: false
    default: "1.9.0"
    description: "OpenTofu version to install"
  oras-registry:
    required: false
    default: ""
    description: "OCI registry prefix for state artifacts (e.g. ghcr.io/your-org/repo-state). Empty = local state only."
  github-token:
    required: true
    description: "Token with admin scope on the target repo (GitHub App install token preferred)"

outputs:
  blueprint-path:
    description: "Path to the generated blueprint.auto.tfvars.json"
    value: ${{ steps.merge.outputs.blueprint-path }}
  stripped-keys:
    description: "JSON array of override keys that were stripped because they conflict with enforced policy"
    value: ${{ steps.merge.outputs.stripped-keys }}
  plan-summary:

runs:
  using: composite
  steps:

      id: paths
      shell: bash
      run: |
        ACTION_DIR="${{ github.action_path }}"
        ENFORCED="${{ inputs.enforced-path }}"
        DEFAULTS="${{ inputs.defaults-path }}"
        TOFU_DIR="${{ inputs.tofu-dir }}"

        echo "enforced=${ENFORCED:-${ACTION_DIR}/config/enforced.yml}" >> "$GITHUB_OUTPUT"
        echo "defaults=${DEFAULTS:-${ACTION_DIR}/config/defaults.yml}" >> "$GITHUB_OUTPUT"
        echo "tofu-dir=${TOFU_DIR:-${ACTION_DIR}/tofu}" >> "$GITHUB_OUTPUT"
        echo "build-dir=${RUNNER_TEMP}/gh-settings-build" >> "$GITHUB_OUTPUT"

            repoSettings: process.env.INPUT_REPO_SETTINGS,
            buildDir: process.env.INPUT_BUILD_DIR,
          }})

    - name: Pull state from ORAS
      if: inputs.oras-registry != ''
      shell: bash
      env:
        ORAS_REGISTRY: ${{ inputs.oras-registry }}
        TOFU_DIR: ${{ steps.paths.outputs.tofu-dir }}
        GITHUB_TOKEN: ${{ inputs.github-token }}
      run: |
        set -euo pipefail
        REPO_NAME="${GITHUB_REPOSITORY##*/}"
        REF="${ORAS_REGISTRY}/${REPO_NAME}:latest"

        cd "${TOFU_DIR}"
        echo "${GITHUB_TOKEN}" | oras login "${ORAS_REGISTRY%%/*}" -u "${GITHUB_ACTOR}" --password-stdin

          echo "Pulled state from ${REF}"
        else
          echo "No existing state at ${REF}, starting fresh"
        fi

    - name: OpenTofu init + plan/apply
      id: tofu
      shell: bash
      env:
        MODE: ${{ inputs.mode }}
        TOFU_DIR: ${{ steps.paths.outputs.tofu-dir }}
        BUILD_DIR: ${{ steps.paths.outputs.build-dir }}
        GITHUB_TOKEN: ${{ inputs.github-token }}
        GITHUB_OWNER: ${{ github.repository_owner }}

# gh-settings

Merge org-enforced + default GitHub settings with per-repo overrides, then
`plan` or `apply` via OpenTofu.

## Model

Three layers, merged in this order:

1. `config/defaults.yml` — org baseline (bundled with this action).
2. `.github/repo-settings.yml` — per-repo overrides (in the calling repo).
3. `config/enforced.yml` — org-wide non-overrideable policy (bundled).

```
stripped_local = repo_settings - enforced_keys
merged         = deep_merge(defaults, stripped_local)
final          = deep_merge(merged, enforced)
```

Enforced keys present in a repo override are **stripped** before merging and
surfaced as a warning + PR-comment notice. They are then reapplied last, so
policy cannot be weakened by a project repo.

## Distribution model

- **Central `github-management` repo** owns `enforced.yml`, `defaults.yml`,
  the merge script, and the OpenTofu module. It publishes this action under
  a tag (e.g. `@v1`).
- **Project repos** call the published action from a small workflow. They
  provide only their `.github/repo-settings.yml` and the trigger.
- **State** is stored per-repo in an OCI registry (GHCR) via ORAS. Each repo
  gets its own artifact tag: `ghcr.io/<org>/repo-state/<repo>:latest`.

## Usage (project repo)

```yaml
# .github/workflows/github-settings.yml
```

## Usage (local iteration in astro-bay)

While developing the action inside this repo, call it with a local path:

```yaml
- uses: ./.github/actions/gh-settings
  with:
    mode: plan
    github-token: ${{ secrets.GH_SETTINGS_TOKEN }}
```

## Files

- `action.yml` — composite action definition.
- `merge.js` — builds `blueprint.auto.tfvars.json` from the three layers.
- `comment.js` — upserts a PR comment with the plan output.
- `config/enforced.yml` — example enforced policy (replace with yours).
- `config/defaults.yml` — example defaults (replace with yours).
- `tofu/` — OpenTofu module that consumes the blueprint.
- `package.json` — declares `js-yaml` as the one runtime dep for the merge
  script. `actions/github-script@v7` runs the script in a Node context that
  already has `@actions/*` and Octokit available; `js-yaml` is the only
  thing we need to install explicitly at action time.

## Installing the js-yaml dep at action time

`actions/github-script` does not auto-install arbitrary npm deps. For now
the merge script uses `js-yaml` which is already present in the runner's
`actions/github-script` bundle. If that changes, add a step that runs
`npm install --prefix "${{ github.action_path }}" --no-audit --no-fund`
before the merge step.

# .github/actions/git-stage-artifacts/action.yml
name: Stage Artifacts
description: >
  Download patterned workflow artifacts and copy files into
  target directories based on a provided mapping.

inputs:
  artifact-prefix:
    required: true
    description: "Artifact name prefix to match (e.g. fuzz-regressions). Each artifact is expected to be named <prefix>-<id>."
  targets:
    required: true
    description: "JSON object mapping artifact IDs to target directory paths (e.g. {\"engine-fuzz\": \"engine/fuzz/regressions\"})"

outputs:
  file-count:
    description: "Total number of files staged"
    value: ${{ steps.stage.outputs.file-count }}
  projects:
    description: "Comma-separated list of IDs that had files staged"
    value: ${{ steps.stage.outputs.projects }}

runs:
  using: composite
  steps:
    - uses: actions/download-artifact@v4
      with:
        pattern: ${{ inputs.artifact-prefix }}-*
        path: /tmp/staged-artifacts

    - id: stage
      shell: bash
      run: |
        shopt -s nullglob
        TARGETS='${{ inputs.targets }}'
        PREFIX="${{ inputs.artifact-prefix }}"
        FILECOUNT=0
        PROJECTS=""

        for dir in /tmp/staged-artifacts/${PREFIX}-*/; do
          ID=$(basename "$dir" | sed "s/^${PREFIX}-//")
          TARGET=$(echo "$TARGETS" | jq -r --arg id "$ID" '.[$id] // empty')

          if [[ -z "$TARGET" ]]; then continue; fi

          COUNT=$(find "$dir" -type f | wc -l)
          if [[ "$COUNT" -eq 0 ]]; then continue; fi

          mkdir -p "$TARGET"
          cp -n "$dir"/* "$TARGET/"
          FILECOUNT=$((FILECOUNT + COUNT))
          PROJECTS="${PROJECTS:+${PROJECTS}, }${ID}"
        done

        echo "file-count=${FILECOUNT}" >> "$GITHUB_OUTPUT"
        echo "projects=${PROJECTS}" >> "$GITHUB_OUTPUT"

        rm -rf /tmp/staged-artifacts

########## FILE: configs/github/.github/actions/moon-raa/README.md ##########
# moon-raa

Manage review apps and artifacts (RAA) for a Moon monorepo. Creates GitHub Deployments, runs per-project build/teardown tasks, and produces a manifest of artifacts.

## Usage

```yaml
- uses: ./.github/actions/moon-raa
  with:
    command: run # or rm
    pr-number: ${{ inputs.pr-number }}
    github-token: ${{ secrets.GITHUB_TOKEN }}
```

## Inputs

| Input          | Required | Default | Description                                               |
| -------------- | -------- | ------- | --------------------------------------------------------- |
| `command`      | yes      |         | `run` to build review artifacts, `rm` to tear down        |
| `pr-number`    | yes      |         | PR number                                                 |
| `prefix`       | no       | `raa`   | Environment name prefix (produces `<prefix>-pr-<number>`) |
| `github-token` | yes      |         | `GITHUB_TOKEN` for deployment API calls                   |

## Outputs

| Output          | Command | Description                                   |
| --------------- | ------- | --------------------------------------------- |
| `deployment-id` | run     | GitHub Deployment ID                          |
| `environment`   | both    | Environment name (e.g. `raa-pr-42`)           |
| `project-count` | both    | Number of projects processed                  |
| `summary`       | both    | Markdown summary for `GITHUB_STEP_SUMMARY`    |
| `manifest`      | run     | Full payload as JSON                          |
| `clean`         | rm      | `true` if teardown completed without warnings |

## Project Configuration

The action discovers affected projects via `moon query projects --affected` and filters to those with `raa` metadata in their `moon.yml`.

Both `run` and `rm` fields are required. Projects missing either field are skipped with a warning.

```yaml
# plugins/my-plugin/moon.yml
project:
  metadata:
    raa:
      run: review # task to run for review build
      rm: review-rm # task to run for teardown
```

The action executes `moon run <project-id>:<raa.run>` during `command: run` and `moon run <project-id>:<raa.rm>` during `command: rm`.

## Workflow Integration

RAA is dispatched from `pr.yml` via `raa.yml`:

1. PR push triggers `pr.yml`, which runs full CI.
2. After the gate passes, if the `review-app` label is present, `pr.yml` dispatches `raa.yml` with `command: run`.
3. On PR close, `pr.yml` dispatches `raa.yml` with `command: rm`.
4. `raa.yml` can also be triggered manually from the Actions tab.

## How It Works

**`run` command:**

1. Discovers affected projects with valid `raa` metadata.
2. Deactivates any existing deployments for the environment.
3. Creates a new GitHub Deployment.
4. Runs each project's `raa.run` task sequentially.
5. Sets deployment status to `success` or `failure`.

**`rm` command:**

1. Reads the deployment manifest to find projects.
2. Runs each project's `raa.rm` task.
3. Deactivates and deletes the deployment.
4. Deletes the GitHub environment.

          gh pr edit "$EXISTING" --title "${{ inputs.title }}" --body "${{ inputs.body }}"
          echo "number=${EXISTING}" >> "$GITHUB_OUTPUT"
        else
          git checkout -b "$BRANCH"
          git commit -m "${{ inputs.title }}"
          git push origin "$BRANCH"
          PR_NUMBER=$(gh pr create \
            --title "${{ inputs.title }}" \
            --body "${{ inputs.body }}" \
            --head "$BRANCH" \
            | grep -oP '#\K[0-9]+' || gh pr list --head "$BRANCH" --json number -q '.[0].number')
          echo "number=${PR_NUMBER}" >> "$GITHUB_OUTPUT"
        fi

            state: '${{ inputs.state }}',
            context: '${{ inputs.context }}',
            description: '${{ inputs.description }}',
            target_url: runUrl,

            ...context.repo,

          const fullBody = marker + '\n' + body;
          if (existing) {
            await github.rest.issues.updateComment({
              ...context.repo, comment_id: existing.id, body: fullBody,
            });
          } else {
            await github.rest.issues.createComment({
              ...context.repo, issue_number: pr, body: fullBody,
            });

      - uses: ./.github/actions/fuzz-cache
        with:
          mode: pull
          project-id: ${{ matrix.project.id }}

          if [[ "$EVENT_NAME" == "workflow_dispatch" ]]; then
            if [[ "$INPUT_SNAPSHOT" == "true" ]]; then
              TASK="review"
            fi

            if [[ "$INPUT_SCOPE" == "affected" ]]; then
              TARGET=":${TASK} --affected"
            else
              TARGET=":${TASK}"
            fi
          fi

          if [[ "$EVENT_NAME" == "release" ]]; then
            COMP=$(echo "$RELEASE_TAG" | sed -E 's/-v[0-9].*//')

            if [[ "$RELEASE_PRERELEASE" == "true" ]]; then
              TASK="review"
            else
              TASK="publish"
              ENV="release"
            fi

            TARGET="${COMP}:${TASK}"
          fi

          {
            echo "task=${TASK}"
            echo "target=${TARGET}"
            echo "environment=${ENV}"
          } >> "$GITHUB_OUTPUT"
        env:
          EVENT_NAME: ${{ github.event_name }}
          INPUT_SNAPSHOT: ${{ inputs.snapshot }}
          INPUT_SCOPE: ${{ inputs.scope }}
          RELEASE_TAG: ${{ github.event.release.tag_name }}
          RELEASE_PRERELEASE: ${{ github.event.release.prerelease }}
  publish:
    name: Moon Run ${{ needs.router.outputs.target }}
    needs: router
    runs-on: ubuntu-latest
    timeout-minutes: 15
    environment: ${{ needs.router.outputs.environment }}
    permissions:
      contents: write
      packages: write
      id-token: write
    steps:
      # Shallow checkout so GH Actions can resolve local composite actions.
      # The setup action re-checks out with full history for moon's affected detection.
      # TODO: Remove once composite actions are published to a shared repo.
      - uses: actions/checkout@v4
        with:
          fetch-depth: 1
      - uses: ./.github/actions/setup

      - name: Execute Moon Task
        run: moon run ${{ needs.router.outputs.target }}
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          CARGO_REGISTRY_TOKEN: ${{ secrets.CARGO_REGISTRY_TOKEN }}
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
          NPM_CONFIG_PROVENANCE: "true"

      - name: Attach standalone artifacts to release
        if: github.event_name == 'release'
        run: |
          if [ -d dist/ ]; then
            gh release upload "${{ github.event.release.tag_name }}" dist/* --clobber
          fi
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}

########## END: configs/github/.github/workflows/publish.yml ##########

name: Review App & Artifacts

on:
  workflow_dispatch:
    inputs:
      command:
        description: run or rm
        type: choice
        options:
          - run
          - rm
        required: true
      pr-number:
        description: PR number
        type: string
        required: true

env:
  FORCE_JAVASCRIPT_ACTIONS_TO_NODE24: true
  PR_NUMBER: ${{ inputs.pr-number }}
  NODE_AUTH_TOKEN: ${{ secrets.GITHUB_TOKEN }}

concurrency:
  group: raa-${{ inputs.pr-number }}
  cancel-in-progress: true

jobs:
  raa:
    name: RAA ${{ inputs.command }} PR #${{ inputs.pr-number }}
    runs-on: ubuntu-latest
    timeout-minutes: 15
    permissions:
      deployments: write
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: ./.github/actions/setup

      - id: raa
        uses: ./.github/actions/moon-raa
        with:
          command: ${{ inputs.command }}
          pr-number: ${{ inputs.pr-number }}
          github-token: ${{ secrets.GITHUB_TOKEN }}

      - name: Summary
        if: always()
        run: echo "$SUMMARY" >> "$GITHUB_STEP_SUMMARY"
        env:
          SUMMARY: ${{ steps.raa.outputs.summary }}

########## END: configs/github/.github/workflows/raa.yml ##########
